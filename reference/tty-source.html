 593 /// Publish a waiter and, when `register` returns true, mark the task Blocked
 594 /// in one interrupt-disabled critical section. The callback runs with SCHEDULER
 595 /// locked, so it must not yield or reacquire that lock. This function does not
 596 /// switch tasks; the caller decides when to yield and how to protect that gap.
 597 pub fn block_current_if(register: impl FnOnce(Pid) -> bool) -> bool {
 598     x86_64::instructions::interrupts::without_interrupts(|| {
 599         let mut sched = SCHEDULER.lock();
 600         let current = sched.current;
 601         let pid = sched.tasks[current].pid;
 602         if register(pid) {
 603             sched.tasks[current].state = TaskState::Blocked;
 604             true
 605         } else {
 606             false
 607         }
 608     })
 609 }

 611 /// Change a matching Blocked task to Ready. It runs on a later scheduling
 612 /// decision. Missing PIDs and tasks in other states are left unchanged.
 613 pub fn wake(pid: Pid) {
 614     x86_64::instructions::interrupts::without_interrupts(|| {
 615         if let Some(task) = SCHEDULER
 616             .lock()
 617             .tasks
 618             .iter_mut()
 619             .find(|task| task.pid == pid)
 620         {
 621             if task.state == TaskState::Blocked {
 622                 task.state = TaskState::Ready;
 623             }
 624         }
 625     });

  25     /// Read at most one queued byte. Receive error bits are discarded.
  26     pub fn try_read_byte(&mut self) -> Option<u8> {
  27         unsafe {
  28             if core::ptr::read_volatile(self.base.add(UART_FLAGS_WORD)) & RX_FIFO_EMPTY != 0 {

  26 /// Dispatch Gabix's small ARM ABI (its numbers are defined above).
  27 /// Fixed scheduled processes can exit/getpid but cannot reboot the machine.
  28 /// Bootstrap user programs use the platform-specific exit/reboot path.
  29 /// Safety: frame must be the live SVC frame, with valid current-process state.
  30 pub unsafe fn dispatch(frame: &mut UserFrame) {
  31     if let Some(id) = crate::scheduler::current_process() {
  32         match frame.registers[SYSCALL_NUMBER_REGISTER] {
  33             SYS_EXIT => crate::process::exit_current(frame.registers[0]),
  34             SYS_GETPID => {
  35                 frame.registers[RESULT_REGISTER] = id as u32 + 1;
  36                 return;
  37             }
  38             SYS_REBOOT => {
  39                 frame.registers[RESULT_REGISTER] = SYSCALL_ERROR;
  40                 return;
  41             }
  42             _ => {}
  43         }
  44     }
  45     let result = match frame.registers[SYSCALL_NUMBER_REGISTER] {
  46         SYS_WRITE => write(frame.registers[0], frame.registers[1], frame.registers[2]),
  47         SYS_EXIT => exit(frame),
  48         SYS_REBOOT => crate::platform::reset(),
  49         _ => SYSCALL_ERROR,
  50     };
  51     frame.registers[RESULT_REGISTER] = result;
  52 }
